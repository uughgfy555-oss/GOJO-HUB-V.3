local function GetNearestFridgePrompt()
    local char = LocalPlayer.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") then return nil end
    local hrp = char.HumanoidRootPart

    local nearestPrompt = nil
    local minDistance = 12

    for _, prompt in pairs(workspace:GetDescendants()) do
        if prompt:IsA("ProximityPrompt") then
            local parentPart = prompt.Parent
            if parentPart and parentPart:IsA("BasePart") then
                local distance = (hrp.Position - parentPart.Position).Magnitude
                if distance < minDistance then
                    local parentName = parentPart.Name
                    local grandParentName = parentPart.Parent and parentPart.Parent.Name or ""
                    
                    if parentName == "Fridge" or prompt.Name:lower():find("fridge") or grandParentName == "Fridge" or prompt.ObjectText == "ซ่อม" then
                        minDistance = distance
                        nearestPrompt = prompt
                    end
                end
            end
        end
    end
    return nearestPrompt
end

local function DoSpamSingleFridge(prompt)
    prompt.RequiresLineOfSight = false
    StatusLabel.Text = "Working..."

    local duration = prompt.HoldDuration > 0 and prompt.HoldDuration or 1.2
    local startTime = tick()

    while (tick() - startTime) < duration and AutoFarmEnabled do
        fireproximityprompt(prompt)
        task.wait(SPAM_INTERVAL)
    end

    StatusLabel.Text = "Done!"
end

FarmToggleBtn.MouseButton1Click:Connect(function()
    AutoFarmEnabled = not AutoFarmEnabled
    if AutoFarmEnabled then
        FarmToggleBtn.Text = "ฟรามเทพ: ON"
        FarmToggleBtn.TextColor3 = Color3.fromRGB(50, 255, 50)
        FrameStroke.Color = Color3.fromRGB(50, 255, 50)
    else
        FarmToggleBtn.Text = "ฟรามเทพ: OFF"
        FarmToggleBtn.TextColor3 = Color3.fromRGB(255, 50, 50)
        FrameStroke.Color = Color3.fromRGB(255, 50, 50)
        StatusLabel.Text = "Status: OFF"
    end
end)

task.spawn(function()
    while true do
        task.wait(0.1)
        if AutoFarmEnabled then
            local prompt = GetNearestFridgePrompt()
            if prompt then
                DoSpamSingleFridge(prompt)
            else
                StatusLabel.Text = "No Fridge Nearby"
            end
        end
    end
end)
